import os

def empaquetar_proyecto(directorio_raiz, archivo_salida):
    # Detección estricta: Si la ruta CONTIENE alguna de estas carpetas, se ignora todo.
    carpetas_prohibidas = [
        'node_modules', 'venv', 'env', '__pycache__', '.git', 
        '.vscode', '.idea', 'build', 'dist', 'postgres_data', 
        'datos_listos_dify', 'public'
    ]
    
    # Extensiones de archivos que NO son código fuente
    excluir_exts = {'.png', '.jpg', '.jpeg', '.gif', '.ico', '.pdf', '.zip', '.mp4', '.sqlite3', '.pyc', '.md', '.xlsx', '.exe', '.lock'}

    print(f"Empaquetando código desde: {os.path.abspath(directorio_raiz)}")
    
    with open(archivo_salida, 'w', encoding='utf-8') as f_out:
        for raiz, dirs, archivos in os.walk(directorio_raiz):
            # Filtro agresivo: Elimina las carpetas prohibidas antes de entrar
            dirs[:] = [d for d in dirs if d not in carpetas_prohibidas]
            
            # Filtro secundario: Si la ruta actual por alguna razón se coló (ej. subcarpetas)
            ruta_separada = os.path.normpath(raiz).split(os.sep)
            if any(prohibida in ruta_separada for prohibida in carpetas_prohibidas):
                continue
                
            for archivo in archivos:
                # Evitar extensiones no deseadas, locks, logs, o el archivo de salida
                if any(archivo.endswith(ext) for ext in excluir_exts):
                    continue
                if archivo.endswith('package-lock.json') or archivo == os.path.basename(archivo_salida):
                    continue
                
                ruta_completa = os.path.join(raiz, archivo)
                
                f_out.write(f"\n{'='*80}\n")
                f_out.write(f"ARCHIVO: {ruta_completa}\n")
                f_out.write(f"{'='*80}\n\n")
                
                try:
                    with open(ruta_completa, 'r', encoding='utf-8') as f_in:
                        f_out.write(f_in.read())
                except Exception as e:
                    f_out.write(f"[Error leyendo archivo: {e}]\n")

if __name__ == '__main__':
    empaquetar_proyecto('.', 'codigo_consolidado.txt')
    print("Proyecto empaquetado exitosamente en 'codigo_consolidado.txt'")

# ==============================================================================
# Orquestador de Micro-Lotes para Bypass de Rate Limit (API de Google - 100 RPM)
# Descripción: Lee el documento PQR extremo y lo divide matemáticamente en 
# lotes de seguridad inferiores a 70 KB, asegurando que la concurrencia de Dify 
# jamás dispare el error HTTP 429 de RESOURCE_EXHAUSTED.
# ==============================================================================

import os
import sys

def generar_micro_lotes_seguros(ruta_archivo_origen: str, directorio_salida: str, limite_kb_seguro: int = 70):
    """
    Fracciona un archivo Markdown masivo en partes más pequeñas (micro-lotes),
    respetando los bloques semánticos definidos por "# --- DOCUMENTO FUENTE".
    """
    # 1. Validación estricta de rutas operativas
    if not os.path.exists(ruta_archivo_origen):
        print(f"[ERROR CRÍTICO] No se localizó el archivo fuente en la ruta: {ruta_archivo_origen}")
        sys.exit(1)

    if not os.path.exists(directorio_salida):
        os.makedirs(directorio_salida)
        print(f"[AUDITORÍA] Directorio de salida creado en: {directorio_salida}")

    limite_bytes_seguro = limite_kb_seguro * 1024
    
    # 2. Lectura y extracción del contenido en memoria
    try:
        with open(ruta_archivo_origen, "r", encoding="utf-8") as f:
            contenido_total = f.read()
    except Exception as e:
        print(f"[ERROR CRÍTICO] Fallo en la decodificación UTF-8 del archivo fuente: {str(e)}")
        sys.exit(1)

    # 3. Segmentación algorítmica por delimitador semántico
    delimitador_principal = "# --- DOCUMENTO FUENTE:"
    
    # Se utiliza un particionamiento que preserva el contexto del separador
    bloques_crudos = contenido_total.split(delimitador_principal)
    
    # El primer bloque puede estar vacío o contener cabeceras irrelevantes antes del primer documento
    bloques_validados = []
    for bloque in bloques_crudos:
        if bloque.strip():
            # Se reconstruye la integridad estructural del documento
            bloques_validados.append(delimitador_principal + bloque)

    total_bloques_extraidos = len(bloques_validados)
    print(f"[INFO] Análisis léxico completado: {total_bloques_extraidos} documentos individuales detectados en el archivo maestro.")

    # 4. Ensamblaje de Micro-Lotes (Batching)
    numero_lote_actual = 1
    peso_acumulado_lote_actual_bytes = 0
    buffer_lote = []
    lotes_ensamblados = 0

    for bloque in bloques_validados:
        peso_bloque_individual_bytes = len(bloque.encode('utf-8'))
        
        # Validación de sobrecarga: Si el bloque actual rompe el límite del lote, se procede al volcado en disco
        if peso_acumulado_lote_actual_bytes + peso_bloque_individual_bytes > limite_bytes_seguro and buffer_lote:
            _escribir_lote_en_disco(buffer_lote, numero_lote_actual, directorio_salida)
            lotes_ensamblados += 1
            
            # Reseteo de variables de estado para el siguiente lote
            numero_lote_actual += 1
            buffer_lote = []
            peso_acumulado_lote_actual_bytes = 0
            
        # Adición del bloque al buffer de RAM
        buffer_lote.append(bloque)
        peso_acumulado_lote_actual_bytes += peso_bloque_individual_bytes

    # 5. Volcado residual: Escribir cualquier bloque remanente que no alcanzó a activar el límite
    if buffer_lote:
        _escribir_lote_en_disco(buffer_lote, numero_lote_actual, directorio_salida)
        lotes_ensamblados += 1

    print(f"\n[ESTADO FINAL] Particionamiento culminado exitosamente.")
    print(f"[REPORTE] Se generaron {lotes_ensamblados} archivos de micro-lotes en la carpeta '{directorio_salida}'.")
    print(f"[DIRECTRIZ] Cada archivo está matemáticamente blindado para no exceder las 100 peticiones por minuto en Dify.")

def _escribir_lote_en_disco(buffer_datos: list, indice_lote: int, ruta_destino: str):
    """
    Función de bajo nivel para ejecutar la persistencia de datos en el sistema de archivos.
    """
    nombre_archivo = f"lote_seguro_pqr_parte_{indice_lote}.md"
    ruta_completa = os.path.join(ruta_destino, nombre_archivo)
    
    contenido_ensamblado = "".join(buffer_datos)
    peso_kb = len(contenido_ensamblado.encode('utf-8')) / 1024
    
    with open(ruta_completa, "w", encoding="utf-8") as archivo:
        archivo.write(contenido_ensamblado)
        
    print(f"  -> Volcado en disco: {nombre_archivo} | Peso estructural: {peso_kb:.2f} KB")

def agregar_log_global(mensaje: str):
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    log_formateado = f"[{timestamp}] {mensaje}"
    _historial_logs.append(log_formateado)
    if len(_historial_logs) > 200:

    logger.info(f"AUDITORÍA SCRA: {mensaje}")

class ScrapingRequest(BaseModel):
    url_objetivo: str
    limite_paginas: int = 15

async def ejecutar_scraping_segundo_plano(job_id: str, url: str, limite: int):
    try:
        agregar_log_global(f"[SISTEMA] Iniciando extracción en URL: {url} (Límite: {limite})")
        paginas_extraidas = await rastrear_dominio(
            url_base=url, 
            limite_paginas=limite, 
            job_id=job_id, 
            dict_estado=trabajos_activos
        )

        if trabajos_activos[job_id]["estado_trabajo"] == "cancelled":
            agregar_log_global("[SISTEMA] Rastreo cancelado por el administrador.")
            return

        if not paginas_extraidas:
            trabajos_activos[job_id]["estado_trabajo"] = "failed"
            agregar_log_global("[ERROR] No se logró extraer contenido válido de la URL.")
            return

        agregar_log_global(f"[SISTEMA] Extracción cruda finalizada. Páginas leídas: {len(paginas_extraidas)}")
        
        texto_crudo = ""
        for i, pag in enumerate(paginas_extraidas):
            texto_crudo += f"INDICE_REGISTRO: {str(i).zfill(4)}\nORIGEN_URL: {pag['url']}\n---\n\n{pag['markdown']}\n\n"

        agregar_log_global("[SISTEMA] Aplicando modelo ML de deduplicación semántica...")
        texto_optimizado = generar_corpus_optimizado(texto_crudo)

        fecha_str = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
        nombre_crudo = f"conocimiento_crudo_{fecha_str}.md"
        nombre_opt = f"conocimiento_optimizado_{fecha_str}.md"
        
        ruta_crudo = os.path.join(DIR_DESCARGAS, nombre_crudo)
        ruta_opt = os.path.join(DIR_DESCARGAS, nombre_opt)

        with open(ruta_crudo, "w", encoding="utf-8") as f:
            f.write(texto_crudo)
            
        with open(ruta_opt, "w", encoding="utf-8") as f:
            f.write(texto_optimizado)

        agregar_log_global("[ÉXITO] Bases de conocimiento renderizadas y guardadas en disco local.")

        # --- INTEGRACIÓN CON DIFY API ---
        agregar_log_global("[SISTEMA] Iniciando protocolo de sincronización con Dify...")
        await sincronizar_documento_dify(ruta_opt, agregar_log_global)

        trabajos_activos[job_id].update({
            "estado_trabajo": "completed",
            "mensaje": "Proceso integral completado.",
            "archivo_crudo": ruta_crudo,
            "archivo_optimizado": ruta_opt
        })

    except Exception as e:
        agregar_log_global(f"[ERROR CRÍTICO] Fallo general: {str(e)}")
        trabajos_activos[job_id]["estado_trabajo"] = "error"
        trabajos_activos[job_id]["mensaje"] = str(e)

@router.post("/iniciar")
async def iniciar_extraccion(request: ScrapingRequest, background_tasks: BackgroundTasks):
    job_id = str(uuid.uuid4())
    trabajos_activos[job_id] = {
        "estado_trabajo": "procesando",
        "total_paginas": 0,
        "mensaje": "Iniciando rastreo de dominio...",
        "archivo_crudo": None,
        "archivo_optimizado": None
    }
    background_tasks.add_task(ejecutar_scraping_segundo_plano, job_id, request.url_objetivo, request.limite_paginas)
    return {"status": "success", "job_id": job_id, "mensaje": "Tarea iniciada."}

@router.get("/estado/{job_id}")
async def consultar_estado(job_id: str):
    if job_id not in trabajos_activos:
        raise HTTPException(status_code=404, detail="El ID de tarea no existe.")
    return trabajos_activos[job_id]

@router.delete("/cancelar/{job_id}")
async def cancelar_extraccion(job_id: str):
    if job_id not in trabajos_activos:
        raise HTTPException(status_code=404, detail="El ID de tarea no existe.")
    trabajos_activos[job_id]["estado_trabajo"] = "cancelled"
    return {"status": "success", "mensaje": "Señal de cancelación enviada al crawler."}

@router.get("/descargar/{nombre_archivo}")
async def descargar_archivo(nombre_archivo: str):
    ruta_archivo = os.path.join(DIR_DESCARGAS, nombre_archivo)
    if not os.path.exists(ruta_archivo):
        raise HTTPException(status_code=404, detail="El archivo solicitado no existe.")
    return FileResponse(path=ruta_archivo, filename=nombre_archivo, media_type='text/markdown')

@router.get("/historial-archivos")
async def listar_archivos_conocimiento():
    if not os.path.exists(DIR_DESCARGAS):
        return {"archivos": []}
    lista_archivos = []
    for nombre_archivo in os.listdir(DIR_DESCARGAS):
        if nombre_archivo.endswith('.md'):
            ruta_completa = os.path.join(DIR_DESCARGAS, nombre_archivo)
            tamaño_kb = round(os.path.getsize(ruta_completa) / 1024, 1)
            fecha_mod_ts = os.path.getmtime(ruta_completa)
            fecha_str = datetime.fromtimestamp(fecha_mod_ts).strftime("%Y-%m-%d %H:%M:%S")
            tipo = "Optimizado (IA)" if "optimizado" in nombre_archivo.lower() else "Original (Crudo)"
            lista_archivos.append({
                "nombre": nombre_archivo,
                "tipo": tipo,
                "peso_kb": tamaño_kb,
                "fecha": fecha_str,
                "timestamp_raw": fecha_mod_ts
            })
    lista_archivos.sort(key=lambda x: x["timestamp_raw"], reverse=True)
    return {"archivos": lista_archivos}

def _calcular_proxima_ejecucion(frecuencia: str, hora_str: str, dia_semana: str, fecha_inicio_str: str = None) -> datetime:
    ahora = datetime.now()
    base_dt = ahora
    if fecha_inicio_str:
        inicio_req = datetime.strptime(fecha_inicio_str, "%Y-%m-%d")
        if inicio_req > ahora: base_dt = inicio_req
    hora, minuto = map(int, hora_str.split(":"))
    proxima = base_dt.replace(hour=hora, minute=minuto, second=0, microsecond=0)
    if frecuencia == "diario":
        if proxima <= ahora: proxima += timedelta(days=1)
    elif frecuencia == "semanal":
        dia_objetivo = DIAS_MAP.get(dia_semana, 0)
        dias_faltantes = (dia_objetivo - proxima.weekday()) % 7
        proxima += timedelta(days=dias_faltantes)
        if proxima <= ahora: proxima += timedelta(days=7)
    return proxima

async def _bucle_cron_nativo():
    agregar_log_global("[CRON] Hilo de automatización activado en segundo plano.")
    while _cron_config["programado"]:
        ahora = datetime.now()
        if _cron_config["fecha_fin"]:
            limite_fin = datetime.strptime(_cron_config["fecha_fin"], "%Y-%m-%d").replace(hour=23, minute=59, second=59)
            if ahora > limite_fin:
                agregar_log_global("[CRON] Fecha de finalización alcanzada. Tarea desactivada.")
                _cron_config["programado"] = False
                break
        proxima = _calcular_proxima_ejecucion(
            _cron_config["frecuencia"], _cron_config["hora"], _cron_config["dia_semana"], _cron_config["fecha_inicio"]
        )
        _cron_config["siguiente_ejecucion"] = proxima.strftime("%Y-%m-%d %H:%M:%S")
        segundos_espera = max(1.0, (proxima - datetime.now()).total_seconds())
        
        try:
            await asyncio.sleep(segundos_espera)
            if not _cron_config["programado"]: break
            
            agregar_log_global(f"[CRON] Disparando extracción programada en URL: {_cron_config['url_objetivo']}")
            
            # Lanzamos la tarea a nuestro propio endpoint interno
            async with httpx.AsyncClient() as client:
                payload = {
                    "url_objetivo": _cron_config["url_objetivo"],
                    "limite_paginas": _cron_config["limite_paginas"]
                }
                await client.post("http://localhost:8000/api/v1/admin/conocimiento/iniciar", json=payload, timeout=10.0)
            
            await asyncio.sleep(65)
        except asyncio.CancelledError:
            agregar_log_global("[CRON] Tarea cancelada manualmente.")
            break
        except Exception as e:
            logger.error(f"[CRON NATIVO] Error: {e}")
            await asyncio.sleep(60)

from sqlalchemy import func, and_
from typing import Optional

class EstadisticasResponse(BaseModel):
    sesiones_globales: int
    total_preguntas: int
    resueltas_ia: int
    escaladas_humano: int
    tasa_contencion: float
    datos_crudos: List[Dict[str, Any]]
    distribucion_tipologia_sede: List[Dict[str, Any]]

@router.get("/estadisticas", response_model=EstadisticasResponse, status_code=status.HTTP_200_OK)
async def obtener_estadisticas_directivas(
    fecha_inicio: Optional[str] = None, 
    fecha_fin: Optional[str] = None, 
    db: AsyncSession = Depends(get_db)
):
    """
    Retorna la sábana de datos crudos y cruces analíticos pre-procesados para el BI Frontend.
    """
    filtros_sesion = []
    filtros_mensaje = [Mensaje.emisor == "usuario"]

    if fecha_inicio and fecha_fin:
        try:
            inicio_dt = datetime.strptime(f"{fecha_inicio} 00:00:00", "%Y-%m-%d %H:%M:%S")
            fin_dt = datetime.strptime(f"{fecha_fin} 23:59:59", "%Y-%m-%d %H:%M:%S")
            filtros_sesion.append(Sesion.creado_en.between(inicio_dt, fin_dt))
            filtros_mensaje.append(Mensaje.creado_en.between(inicio_dt, fin_dt))
        except ValueError:
            raise HTTPException(status_code=400, detail="Formato de fecha inválido. Use YYYY-MM-DD.")

    # 1. KPIs Globales
    stmt_sesiones = select(func.count(Sesion.id)).where(and_(*filtros_sesion))
    res_sesiones = await db.execute(stmt_sesiones)
    total_sesiones = res_sesiones.scalar() or 0

    stmt_preguntas = select(func.count(Mensaje.id)).where(and_(*filtros_mensaje))
    res_preguntas = await db.execute(stmt_preguntas)
    total_preguntas = res_preguntas.scalar() or 0

    filtros_escaladas = filtros_sesion.copy()
    filtros_escaladas.append(Sesion.estado.in_(["en_cola", "atencion_humana"]))
    stmt_escaladas = select(func.count(Sesion.id)).where(and_(*filtros_escaladas))
    res_escaladas = await db.execute(stmt_escaladas)
    escaladas = res_escaladas.scalar() or 0

    resueltas_ia = total_sesiones - escaladas
    resueltas_ia = max(resueltas_ia, 0)
    tasa = round((resueltas_ia / total_sesiones) * 100, 1) if total_sesiones > 0 else 0.0

    # 2. Datos Crudos Básicos
    stmt_crudos = select(Sesion).where(and_(*filtros_sesion)).order_by(Sesion.creado_en.desc())
    sesiones_crudas = (await db.execute(stmt_crudos)).scalars().all()
    
    datos_crudos = []
    
    # 3. Extracción Analítica de Tipología
    for s in sesiones_crudas:
        tipologia_detectada = "No Clasificada"
        
        # Si la sesión está cerrada, buscamos su mensaje de resolución
        if s.estado == "cerrada":
             stmt_cierre = select(Mensaje.contenido).where(and_(Mensaje.sesion_id == s.id, Mensaje.emisor == "sistema", Mensaje.contenido.like("[RESOLUCIÓN]%")))
             res_cierre = await db.execute(stmt_cierre)
             mensaje_resolucion = res_cierre.scalar()
             
             if mensaje_resolucion:
                 # 1. Búsqueda por el nuevo formato estructurado "|| TIPOLOGIA: valor ||"
                 match_nuevo = re.search(r"\|\|\s*TIPOLOGIA:\s*([^|]+)", mensaje_resolucion)
                 if match_nuevo:
                     tipologia_detectada = match_nuevo.group(1).strip().capitalize()
                 else:
                     # 2. Búsqueda de respaldo (Fallback) para tickets antiguos usando la lógica original
                     # Extrae palabras clave directamente del texto
                     texto_min = mensaje_resolucion.lower()
                     if "peticion" in texto_min or "petición" in texto_min: tipologia_detectada = "Peticion"
                     elif "queja" in texto_min: tipologia_detectada = "Queja"
                     elif "reclamo" in texto_min: tipologia_detectada = "Reclamo"
                     elif "sugerencia" in texto_min: tipologia_detectada = "Sugerencia"
                     elif "felicitacion" in texto_min or "felicitación" in texto_min: tipologia_detectada = "Felicitacion"

        datos_crudos.append({
            "ID_Caso": str(s.id),
            "Fecha_Ingreso": s.creado_en.strftime("%Y-%m-%d %H:%M:%S") if s.creado_en else "",
            "Fecha_Actualizacion": s.actualizado_en.strftime("%Y-%m-%d %H:%M:%S") if s.actualizado_en else "",
            "Nombre_Usuario": s.nombre_usuario or "Anónimo",
            "Documento": s.documento_identidad or "N/A",
            "Contacto_Email": s.correo or "N/A",
            "Contacto_Telefono": s.celular or "N/A",
            "Sede": s.sede or "No Definida",
            "Perfil": s.tipo_usuario or "No Definido",
            "Estado_Final": s.estado.replace("_", " ").title(),
            "Tipologia_PQRSD": tipologia_detectada
        })

    # 4. Agrupación Preparada para Gráfico de Barras Apiladas (Recharts)
    mapa_sedes = {}
    for d in datos_crudos:
        if d["Estado_Final"] == "Cerrada" and d["Tipologia_PQRSD"] != "No Clasificada":
            sede = d["Sede"]
            tipo = d["Tipologia_PQRSD"]
            
            # Normalizamos los tipos para que coincidan exactamente con las llaves que espera Recharts
            if tipo == "Petición": tipo = "Peticion"
            if tipo == "Felicitación": tipo = "Felicitacion"
            
            if sede not in mapa_sedes:
                mapa_sedes[sede] = {"name": sede}
            
            mapa_sedes[sede][tipo] = mapa_sedes[sede].get(tipo, 0) + 1

    distribucion_tipologia_sede = list(mapa_sedes.values())

    return {
        "sesiones_globales": total_sesiones,
        "total_preguntas": total_preguntas,
        "resueltas_ia": resueltas_ia,
        "escaladas_humano": escaladas,
        "tasa_contencion": tasa,
        "datos_crudos": datos_crudos,
        "distribucion_tipologia_sede": distribucion_tipologia_sede
    }

@router.post("/enviar", response_model=dict, status_code=status.HTTP_200_OK)
async def procesar_mensaje_usuario(peticion: PeticionUsuario, db: AsyncSession = Depends(get_db)):
    """
    Recibe un mensaje del usuario, lo guarda, consulta al LLM y evalúa el umbral RAG.
    """
    logger.info(f"Recibiendo petición REST para sesión: {peticion.sesion_id}")
    
    stmt = select(Sesion).where(Sesion.id == peticion.sesion_id)
    resultado = await db.execute(stmt)
    sesion = resultado.scalars().first()

    # Como el chat ahora requiere formulario, si la sesión no existe, rechazamos la petición
    if not sesion:
        raise HTTPException(status_code=403, detail="Sesión no válida o caducada. Por favor, reinicia el chat.")

    if sesion.estado in ["cerrada", "ticket_asincrono"]:
        raise HTTPException(status_code=400, detail="El caso ya ha sido gestionado o cerrado.")

    mensaje_usuario = Mensaje(
        sesion_id=sesion.id,
        emisor="usuario",
        contenido=peticion.texto_pregunta
    )
    db.add(mensaje_usuario)
    await db.commit() 

    # Si la sesión ya fue escalada, el usuario debería usar WebSocket, no REST.
    if sesion.estado in ["en_cola", "atencion_humana"]:
        logger.info(f"Sesión {sesion.id} ignorada por IA. Ya está asignada a humano.")
        return {
            "estado_sesion": sesion.estado,
            "notificacion": "Un asesor institucional ya está atendiendo tu caso. Por favor, usa la ventana de chat en vivo."
        }
    
    # ---------------------------------------------------------
    # INTERCEPTOR DE BOTONES DE ACCIÓN (Evita consultar a Dify)
    # ---------------------------------------------------------
    if peticion.texto_pregunta == "/escalar_humano":
        await _escalar_a_humano(db, sesion.id)
        return {
            "sesion_id": str(sesion.id),
            "estado_actual": "en_cola",
            "respuesta_sistema": "Te he transferido con nuestro equipo de atención institucional. Un asesor se conectará contigo en un momento."
        }
    
    if peticion.texto_pregunta == "/continuar":
        return {
            "sesion_id": str(sesion.id),
            "estado_actual": "atencion_bot",
            "respuesta_sistema": "¡Entendido! Cuéntame, ¿cómo más te puedo ayudar hoy?"
        }
    # ---------------------------------------------------------

    logger.info(f"Enviando consulta a Dify Cloud para sesión: {sesion.id}")
    respuesta_bruta_dify = await consultar_agente_dify(
        pregunta=peticion.texto_pregunta,
        identificador_usuario=str(sesion.id)
    )

    resultado_enrutamiento = await evaluar_y_enrutar_respuesta(
        db=db, 
        sesion_id=sesion.id, 
        respuesta_dify=respuesta_bruta_dify
    )

    # Reestructuración de la respuesta final para admitir el paso de "opciones" (botones) al frontend
    respuesta_final = {
        "sesion_id": str(sesion.id),
        "estado_actual": "en_cola" if resultado_enrutamiento["accion"] == "escalado" else "atencion_bot",
        "respuesta_sistema": resultado_enrutamiento["mensaje_publico"],
        "metadatos": resultado_enrutamiento.get("metadatos", {})
    }

    if "opciones" in resultado_enrutamiento:
        respuesta_final["opciones"] = resultado_enrutamiento["opciones"]

    return respuesta_final

@router.websocket("/ws/{sesion_id}/{rol}")
async def websocket_chat_humano(websocket: WebSocket, sesion_id: str, rol: str):
    await manager.connect(websocket, sesion_id, rol)
    
    try:
        while True:
            datos_recibidos = await websocket.receive_text()
            payload = json.loads(datos_recibidos)
            texto_mensaje = payload.get("contenido", "")
            
            if not texto_mensaje.strip():
                continue

            async with AsyncSessionLocal() as db:
                nuevo_mensaje = Mensaje(
                    sesion_id=sesion_id,
                    emisor=rol, 
                    contenido=texto_mensaje
                )
                db.add(nuevo_mensaje)
                await db.commit()

                stmt = select(Sesion).where(Sesion.id == sesion_id)
                resultado = await db.execute(stmt)
                sesion = resultado.scalars().first()

                if not sesion:
                     continue

                if rol == "asesor":
                    if sesion.estado == 'en_cola':
                         stmt_update = update(Sesion).where(Sesion.id == sesion_id).values(estado="atencion_humana")
                         await db.execute(stmt_update)
                         await db.commit()
                    
                    mensaje_a_emitir = {
                        "tipo": "nuevo_mensaje",
                        "emisor": "asesor",
                        "contenido": texto_mensaje
                    }
                    await manager.broadcast_to_session(sesion_id, mensaje_a_emitir, sender_rol=rol)
                
                elif rol == "usuario":
                    if sesion.estado == "atencion_bot":

import os
import httpx
import logging
import json

logger = logging.getLogger(__name__)

DIFY_URL = os.getenv("DIFY_API_BASE_URL", "https://api.dify.ai/v1")
DATASET_KEY = os.getenv("DIFY_DATASET_API_KEY", "")
DATASET_ID = os.getenv("DIFY_DATASET_ID", "")

async def sincronizar_documento_dify(ruta_archivo: str, log_callback) -> bool:
    """
    Se conecta a la API de Datasets de Dify.
    1. Borra todos los documentos previos del dataset para evitar duplicidad.
    2. Sube el nuevo archivo Markdown con reglas de Alta Calidad (High Quality).
    """
    if not DATASET_KEY or not DATASET_ID:
        log_callback("[DIFY API] IGNORADO: Faltan credenciales DIFY_DATASET_API_KEY o DIFY_DATASET_ID en el entorno.")
        return False

    headers = {"Authorization": f"Bearer {DATASET_KEY}"}

    try:
        async with httpx.AsyncClient(timeout=60.0) as client:
            # PASO 1: Listar documentos actuales en el dataset
            log_callback(f"[DIFY API] Consultando documentos en Dataset ID: {DATASET_ID}")
            res_lista = await client.get(f"{DIFY_URL}/datasets/{DATASET_ID}/documents", headers=headers)
            res_lista.raise_for_status()
            documentos = res_lista.json().get("data", [])

            # PASO 2: Eliminar documentos antiguos (Reemplazo)
            for doc in documentos:
                doc_id = doc.get("id")
                log_callback(f"[DIFY API] Eliminando documento obsoleto: {doc_id}")
                await client.delete(f"{DIFY_URL}/datasets/{DATASET_ID}/documents/{doc_id}", headers=headers)

            # PASO 3: Subir el nuevo documento con configuración híbrida
            log_callback(f"[DIFY API] Preparando subida del nuevo corpus de conocimiento...")
            
            reglas_procesamiento = {
                "indexing_technique": "high_quality",
                "process_rule": {
                    "mode": "custom",
                    "rules": {
                        "pre_processing_rules": [
                            {"id": "remove_extra_spaces", "enabled": True},
                            {"id": "remove_urls_emails", "enabled": False}
                        ],
                        "segmentation": {
                            "separator": "\n\n",
                            "max_tokens": 1000,
                            "chunk_overlap": 150
                        }
                    }
                }
            }

            nombre_archivo = os.path.basename(ruta_archivo)
            
            with open(ruta_archivo, "rb") as f:
                archivos_multipart = {
                    "file": (nombre_archivo, f, "text/markdown")
                }
                datos_form = {
                    "data": json.dumps(reglas_procesamiento)
                }

                res_subida = await client.post(
                    f"{DIFY_URL}/datasets/{DATASET_ID}/document/create_by_file",
                    headers=headers,
                    data=datos_form,
                    files=archivos_multipart
                )
                res_subida.raise_for_status()

            log_callback(f"[DIFY API] ¡ÉXITO! Base de conocimiento sincronizada e indexando en Dify.")
            return True

    except Exception as e:
        log_callback(f"[DIFY API ERROR] Fallo durante la sincronización: {str(e)}")
        logger.error(f"Error Dify Admin: {str(e)}")
        return False

# ==============================================================================
# Cliente API Dify - Agente PQR
# Descripción: Maneja la comunicación HTTP asíncrona real con la nube de Dify.ai
# ==============================================================================

import os
import httpx
import logging
import re
import json

logger = logging.getLogger(__name__)

DIFY_URL = os.getenv("DIFY_API_BASE_URL", "https://api.dify.ai/v1")
DIFY_KEY = os.getenv("DIFY_API_KEY", "")

# Memoria en tiempo de ejecución para preservar el hilo conversacional (sesion_id -> dify_conversation_id)
_mapa_conversaciones_dify = {}

async def consultar_agente_dify(pregunta: str, identificador_usuario: str) -> dict:
    """
    Envía la pregunta del usuario a Dify manteniendo el conversation_id activo,
    limpia las etiquetas de razonamiento y extrae estrictamente la similitud reportada.
    """
    if not DIFY_KEY or "AQUI_PEGA" in DIFY_KEY:
        logger.error("DIFY_API_KEY no es válida o sigue siendo la de prueba.")
        return {"estado": "error", "respuesta": "Error interno de configuración (API Key)."}

    headers = {
        "Authorization": f"Bearer {DIFY_KEY}",
        "Content-Type": "application/json"
    }
    
    # Recuperar el ID de conversación de Dify si esta sesión ya había hablado antes
    conversacion_dify_actual = _mapa_conversaciones_dify.get(identificador_usuario, "")

    payload = {
        "inputs": {},
        "query": pregunta,
        "response_mode": "blocking",
        "conversation_id": conversacion_dify_actual,
        "user": identificador_usuario
    }

    logger.info(f">>> [AUDITORÍA DE RED] Destino HTTP: {DIFY_URL}/chat-messages | Hilo Dify: '{conversacion_dify_actual or 'NUEVO'}'")

    try:
        async with httpx.AsyncClient(timeout=45.0) as client:
            response = await client.post(f"{DIFY_URL}/chat-messages", json=payload, headers=headers)
            
            # Si el conversation_id expiró o cambió de App en Dify, reintentar como hilo nuevo
            if response.status_code == 404 and conversacion_dify_actual:
                logger.warning(">>> [HILO DIFY] Conversación previa no encontrada en Dify. Reiniciando hilo...")
                _mapa_conversaciones_dify.pop(identificador_usuario, None)
                payload["conversation_id"] = ""
                response = await client.post(f"{DIFY_URL}/chat-messages", json=payload, headers=headers)

            response.raise_for_status() 
            
            data = response.json()
            
            # Guardar el conversation_id devuelto por Dify para mantener el hilo en las siguientes preguntas
            nuevo_conv_id = data.get("conversation_id")
            if nuevo_conv_id:
                _mapa_conversaciones_dify[identificador_usuario] = nuevo_conv_id

            texto_respuesta_cruda = data.get("answer", "Lo siento, no pude procesar el texto.")
            
            texto_limpio = re.sub(r'<think>.*?</think>', '', texto_respuesta_cruda, flags=re.DOTALL).strip()
            
            if not texto_limpio:
                texto_limpio = "Lo siento, hubo un error procesando la directriz cognitiva."

            metadata = data.get("metadata", {})
            retriever_resources = metadata.get("retriever_resources", [])
            
            logger.warning(f">>> [AUDITORÍA METADATA DIFY] Recursos recuperados ({len(retriever_resources)}): {json.dumps(retriever_resources, ensure_ascii=False)}")
            
            # Si Dify no envía documentos, la similitud es estrictamente cero.
            similitud_cruda = 0.0 
            url_fuente = "Sin Fuente Recuperada"
            
            if retriever_resources and len(retriever_resources) > 0:
                recursos_ordenados = sorted(
                    retriever_resources, 
                    key=lambda x: float(x.get("score") or 0.0), 
                    reverse=True
                )
                mejor_fragmento = recursos_ordenados[0]
                similitud_cruda = float(mejor_fragmento.get("score") or 0.0)
                url_fuente = mejor_fragmento.get("document_name") or mejor_fragmento.get("dataset_name") or url_fuente

            return {
                "estado": "exito",
                "respuesta": texto_limpio,
                "similitud_rag": similitud_cruda,
                "url_fuente": url_fuente
            }

    except httpx.HTTPStatusError as e:
        logger.error(f"Dify respondió con error HTTP: {e.response.status_code} - {e.response.text}")
        return {"estado": "error", "respuesta": "El servidor de inteligencia artificial está temporalmente fuera de servicio."}
    except Exception as e:
        logger.error(f"Fallo de conexión o timeout con la API de Dify: {str(e)}")

# Cliente API Firecrawl - Módulo de Administración
# Descripción: Maneja la comunicación HTTP asíncrona con Firecrawl para el
# rastreo y extracción de contenido web optimizado para LLMs.
# ==============================================================================

import os
import httpx
import logging
from typing import Dict, Any

logger = logging.getLogger(__name__)

FIRECRAWL_API_URL = "https://api.firecrawl.dev/v1"
FIRECRAWL_API_KEY = os.getenv("FIRECRAWL_API_KEY", "")

async def iniciar_trabajo_crawl(url: str, limite: int) -> Dict[str, Any]:
    """
    Envía una petición POST a Firecrawl para iniciar un trabajo de rastreo profundo.
    Retorna el Job ID para realizar el seguimiento asíncrono.
    """
    if not FIRECRAWL_API_KEY:
        logger.error("FIRECRAWL_API_KEY no está configurada en las variables de entorno.")
        return {"success": False, "error": "API Key no configurada."}

    headers = {
        "Authorization": f"Bearer {FIRECRAWL_API_KEY}",
        "Content-Type": "application/json"
    }

    payload = {
        "url": url,
        "limit": limite,
        "scrapeOptions": {
            "formats": ["markdown"],
            "onlyMainContent": True
        }
    }

    logger.info(f">>> [FIRECRAWL] Iniciando rastreo en URL: {url} con límite de {limite} páginas.")

    try:
        async with httpx.AsyncClient(timeout=30.0) as client:
            response = await client.post(f"{FIRECRAWL_API_URL}/crawl", json=payload, headers=headers)
            response.raise_for_status()
            return response.json()
    except httpx.HTTPStatusError as e:
        logger.error(f"Firecrawl respondió con error HTTP: {e.response.status_code} - {e.response.text}")
        return {"success": False, "error": str(e)}
    except Exception as e:
        logger.error(f"Fallo de conexión con la API de Firecrawl: {str(e)}")
        return {"success": False, "error": str(e)}

async def consultar_estado_crawl(job_id: str) -> Dict[str, Any]:
    """
    Consulta el estado de un trabajo de rastreo previamente iniciado.
    Si el estado es 'completed', el JSON de respuesta contendrá el arreglo 'data' con los markdowns.
    """
    headers = {
        "Authorization": f"Bearer {FIRECRAWL_API_KEY}",
        "Content-Type": "application/json"
    }

    try:
        async with httpx.AsyncClient(timeout=60.0) as client:
            response = await client.get(f"{FIRECRAWL_API_URL}/crawl/{job_id}", headers=headers)
            response.raise_for_status()
            return response.json()
    except Exception as e:
        logger.error(f"Fallo al consultar el estado del trabajo {job_id} en Firecrawl: {str(e)}")
        return {"status": "error", "error": str(e)}

async def cancelar_trabajo_crawl(job_id: str) -> Dict[str, Any]:

    Envía una petición DELETE para abortar un trabajo de rastreo en curso en los servidores de Firecrawl.

        logger.error(f"Fallo al intentar cancelar el trabajo {job_id} en Firecrawl: {str(e)}")

def obtener_umbral_dinamico() -> float:
    try:
        with open("assets/configuracion.json", "r", encoding="utf-8") as f:
            data = json.load(f)
            if "umbral_rag" in data:
                return float(data["umbral_rag"])
    except:

def traducir_distancia_a_confianza(similitud_cruda: float) -> float:
    """
    Convierte la distancia vectorial bruta (0.0 a 1.0) en un porcentaje humano (0% a 100%).
    """
    if similitud_cruda is None:
        return 0.0
    
    sim_float = float(similitud_cruda)
    confianza = sim_float * 100.0
    
    return round(confianza, 1)

def intencion_solicitar_humano(texto_usuario: str) -> bool:
    """
    Verifica si el usuario está pidiendo explícitamente comunicarse con un humano.
    """
    patrones = [
        r"hablar\s+con\s+humano",
        r"hablar\s+con\s+asesor",
        r"comunicar\s+con\s+alguien",
        r"pasar\s+con\s+asesor",
        r"necesito\s+un\s+humano",
        r"hablar\s+con\s+persona"
    ]
    texto_limpio = texto_usuario.lower()
    for patron in patrones:
        if re.search(patron, texto_limpio):
            return True
    return False

async def evaluar_y_enrutar_respuesta(db: AsyncSession, sesion_id: UUID, respuesta_dify: dict = None, peticion_usuario: str = None) -> dict:

        }

# Motor de Extracción Semántica Local (Playwright + BS4 + ML)
# Descripción: Extrae contenido dinámico, limpia el DOM y aplica un aplanador
# semántico mediante Regex para destruir tablas Markdown. Incluye deduplicación
# vectorial para optimización de contexto RAG.
# ==============================================================================

import asyncio
import logging
import re
import numpy as np
from bs4 import BeautifulSoup
from markdownify import markdownify as md
from playwright.async_api import async_playwright
from typing import List, Dict, Any
from urllib.parse import urljoin, urlparse
from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity

logger = logging.getLogger(__name__)

ETIQUETAS_BASURA = [
    'nav', 'footer', 'header', 'aside', 'script', 'style', 
    'noscript', 'iframe', 'svg', 'button', 'form'
]

def aplanar_semantica_markdown(markdown_crudo: str) -> str:
    lineas = markdown_crudo.split('\n')
    resultado = []
    encabezado_actual = ""
    en_tabla = False

    for linea in lineas:
        linea_limpia = linea.strip()

        match_encabezado = re.match(r'^(#{1,6})\s+(.+)', linea_limpia)
        if match_encabezado:
            encabezado_actual = match_encabezado.group(2).strip()
            resultado.append(linea_limpia)
            continue

        match_tabla = re.match(r'^\|(.*)\|$', linea_limpia)
        if match_tabla:
            en_tabla = True
            contenido_filas = match_tabla.group(1)
            
            if "---" in contenido_filas or re.search(r'Sede.*Programa|Programa.*Sede', contenido_filas, re.IGNORECASE):
                continue
            
            columnas = [col.strip() for col in contenido_filas.split('|') if col.strip()]
            
            if len(columnas) == 2:
                col1, col2 = columnas[0], columnas[1]
                if re.search(r'Ingeniería|Licenciatura|Profesional|Administración|Zootecnia|Enfermería|Tecnología|Especialización|Maestría', col1, re.IGNORECASE):
                    oracion = f"El programa de {col1} se oferta en la sede de {col2}."
                else:
                    oracion = f"En la sede de {col1} se oferta el programa de {col2}."
                resultado.append(oracion)
            
            elif len(columnas) == 3:
                programa, sede, fecha = columnas[0], columnas[1], columnas[2]
                resultado.append(f"El programa de {programa} se oferta en la sede de {sede} (Fecha de radicación registrada: {fecha}).")
            
            continue

        if en_tabla and not match_tabla:
            en_tabla = False
            resultado.append("")

        match_lista = re.match(r'^[-*]\s+(.+)', linea_limpia)
        if match_lista:
            item_lista = match_lista.group(1).strip()
            if encabezado_actual:
                resultado.append(f"- [Contexto: {encabezado_actual}] {item_lista}")
            else:
                resultado.append(linea_limpia)
            continue

        resultado.append(linea_limpia)

    texto_final = '\n'.join(resultado)
    texto_final = re.sub(r'\n{3,}', '\n\n', texto_final)
    return texto_final

async def extraer_contenido_pagina(url: str, browser_context) -> Dict[str, Any]:
    page = await browser_context.new_page()
    try:
        await page.goto(url, wait_until="domcontentloaded", timeout=60000)
        
        html_crudo = await page.content()
        titulo = await page.title()
        
        soup = BeautifulSoup(html_crudo, "html.parser")
        
        for elemento in soup.find_all(ETIQUETAS_BASURA):
            elemento.decompose()

        contenedor_principal = soup.find('main') or soup.find('article') or soup.find('div', id=lambda x: x and 'content' in x.lower())
        
        if contenedor_principal:
            html_limpio = str(contenedor_principal)
        else:
            html_limpio = str(soup.body) if soup.body else str(soup)

        markdown_crudo = md(
            html_limpio, 
            heading_style="ATX", 
            strip=['a', 'img'], 
            bullets="-"
        ).strip()
        
        markdown_procesado = aplanar_semantica_markdown(markdown_crudo)

        return {
            "url": url,
            "titulo": titulo,
            "markdown": markdown_procesado,
            "exito": True
        }

    except Exception as e:
        logger.error(f"Fallo al extraer {url}: {str(e)}")
        return {
            "url": url,
            "titulo": "Error",
            "markdown": "",
            "exito": False
        }
    finally:
        await page.close()

async def rastrear_dominio(url_base: str, limite_paginas: int = 50, job_id: str = None, dict_estado: dict = None) -> List[Dict[str, Any]]:
    paginas_extraidas = []
    urls_visitadas = set()
    cola_urls = [url_base]
    dominio_base = urlparse(url_base).netloc

    logger.info(f">>> [SCRAPER LOCAL] Iniciando rastreo profundo en: {url_base}")

    async with async_playwright() as p:
        browser = await p.chromium.launch(headless=True)
        context = await browser.new_context(
            user_agent="Mozilla/5.0 (Windows NT 10.0; Win64; x64) OrquestadorPQR/1.0"
        )

        while cola_urls and len(paginas_extraidas) < limite_paginas:
            if dict_estado and job_id:
                if dict_estado.get(job_id, {}).get("estado_trabajo") == "cancelled":
                    logger.info("Rastreo abortado por solicitud del usuario.")
                    break

            url_actual = cola_urls.pop(0)
            url_actual = url_actual.rstrip('/')
            
            if url_actual in urls_visitadas:
                continue
                
            urls_visitadas.add(url_actual)
            logger.info(f"[{len(paginas_extraidas) + 1}/{limite_paginas}] Extrayendo: {url_actual}")
            
            if dict_estado and job_id:
                dict_estado[job_id]["total_paginas"] = len(paginas_extraidas)

            datos_pagina = await extraer_contenido_pagina(url_actual, context)
            
            if not datos_pagina["exito"] or not datos_pagina["markdown"]:
                continue

            paginas_extraidas.append(datos_pagina)

            page = await context.new_page()
            try:
                await page.goto(url_actual, wait_until="domcontentloaded", timeout=30000)
                enlaces_elementos = await page.query_selector_all("a")
                
                for enlace in enlaces_elementos:
                    href = await enlace.get_attribute("href")
                    if href:
                        url_absoluta = urljoin(url_base, href)
                        url_absoluta = url_absoluta.split('#')[0]
                        
                        if urlparse(url_absoluta).netloc == dominio_base:
                            if not url_absoluta.lower().endswith(('.pdf', '.jpg', '.png', '.zip', '.rar', '.mp4', '.xlsx', '.xls', '.docx', '.doc', '.xlsm')):
                                if url_absoluta not in urls_visitadas and url_absoluta not in cola_urls:
                                    cola_urls.append(url_absoluta)
            except Exception as e:
                logger.warning(f"No se pudieron extraer enlaces hijos de {url_actual}: {str(e)}")
            finally:
                await page.close()

        await browser.close()
        
    logger.info(f">>> [SCRAPER LOCAL] Rastreo finalizado. Páginas extraídas: {len(paginas_extraidas)}")
    return paginas_extraidas

def _limpiar_heuristica_basica(texto: str) -> str:
    texto = re.sub(r'-\s+\[Contexto:.*?\]\s+Inicio\n', '', texto, flags=re.IGNORECASE)
    patrones_cta = [
        r'^Ver más$', r'^Descargar$', r'^Inscríbete$', r'^Leer más$',
        r'^Ver perfil$', r'^Descargar plan rectoral$', r'^Ir a Autoevaluación$',
        r'^Ver más noticias relacionadas$', r'^Ver más información aquí$', r'^Descarga aquí.*$'
    ]
    for patron in patrones_cta:
        texto = re.sub(patron, '', texto, flags=re.MULTILINE | re.IGNORECASE)
    
    texto = re.sub(r'^(?P<linea>.{1,50})$\n+(?P=linea)', r'\g<linea>', texto, flags=re.MULTILINE)
    texto = re.sub(r'\n{3,}', '\n\n', texto)
    return texto.strip()

def generar_corpus_optimizado(texto_completo: str, umbral_similitud: float = 0.88) -> str:
    logger.info(">>> [FILTRO SEMÁNTICO] Iniciando limpieza de ruido RAG...")
    registros = re.split(r'(INDICE_REGISTRO:\s+\d+\nORIGEN_URL:\s+.*?\n---)', texto_completo)
    
    if len(registros) < 3:
        return texto_completo 
        
    logger.info("Cargando modelo Sentence-Transformer en memoria...")
    modelo = SentenceTransformer('paraphrase-multilingual-MiniLM-L12-v2')
    
    documento_optimizado = ""
    total_parrafos_originales = 0
    total_parrafos_retenidos = 0
    
    for i in range(1, len(registros), 2):
        cabecera = registros[i]
        cuerpo = registros[i+1] if i+1 < len(registros) else ""
        
        cuerpo_limpio = _limpiar_heuristica_basica(cuerpo)
        
        bloques_crudos = cuerpo_limpio.split('\n\n')
        parrafos = [bloque.strip() for bloque in bloques_crudos if len(bloque.strip()) > 15]
        total_parrafos_originales += len(parrafos)
        
        if len(parrafos) > 2:
            embeddings = modelo.encode(parrafos, convert_to_tensor=False)
            parrafos_unicos = []
            embeddings_unicos = []
            
            for parrafo, embedding in zip(parrafos, embeddings):
                if not embeddings_unicos:
                    parrafos_unicos.append(parrafo)
                    embeddings_unicos.append(embedding)
                    continue
                
                similitudes = cosine_similarity([embedding], embeddings_unicos)[0]
                if np.max(similitudes) < umbral_similitud:
                    parrafos_unicos.append(parrafo)
                    embeddings_unicos.append(embedding)
            parrafos_optimizados = parrafos_unicos
        else:
            parrafos_optimizados = parrafos
            
        total_parrafos_retenidos += len(parrafos_optimizados)
        cuerpo_final = "\n\n".join(parrafos_optimizados)
        documento_optimizado += f"{cabecera}\n\n{cuerpo_final}\n\n"
        
    logger.info(f">>> [FILTRO SEMÁNTICO] Optimización completada. Párrafos retenidos: {total_parrafos_retenidos} de {total_parrafos_originales}.")
    return documento_optimizado

import json
import logging

logger = logging.getLogger(__name__)

class ConnectionManager:
    def __init__(self):
        # Diccionario en memoria para rastrear las conexiones activas por cada sesión de chat.
        # Estructura esperada: { "sesion_id_uuid": {"estudiante": WebSocket, "asesor": WebSocket} }
        self.active_connections: Dict[str, Dict[str, WebSocket]] = {}

    async def connect(self, websocket: WebSocket, sesion_id: str, rol: str):
        """
        Acepta una nueva conexión WebSocket y la registra en el diccionario en memoria.
        rol debe ser 'estudiante' o 'asesor'.
        """
        await websocket.accept()
        
        if sesion_id not in self.active_connections:
            self.active_connections[sesion_id] = {}
            
        self.active_connections[sesion_id][rol] = websocket
        logger.info(f"Nueva conexión WebSocket: Sesión {sesion_id} - Rol: {rol}")

    def disconnect(self, sesion_id: str, rol: str):
        """
        Elimina la conexión del registro en memoria cuando el usuario cierra la pestaña o pierde red.
        """
        if sesion_id in self.active_connections:
            if rol in self.active_connections[sesion_id]:
                del self.active_connections[sesion_id][rol]
            # Si ambos se desconectan, limpiamos la clave de la sesión para liberar memoria RAM
            if not self.active_connections[sesion_id]:
                del self.active_connections[sesion_id]
        logger.info(f"Desconexión WebSocket: Sesión {sesion_id} - Rol: {rol}")

    async def send_personal_message(self, message: dict, websocket: WebSocket):
        """
        Envía un mensaje JSON de vuelta únicamente al cliente que originó la acción (ej. confirmación de lectura).
        """
        await websocket.send_text(json.dumps(message))

    async def broadcast_to_session(self, sesion_id: str, message: dict, sender_rol: str = None):
        """
        Transmite un mensaje JSON a los participantes conectados en una sesión de chat específica.
        [MODIFICACIÓN ARQUITECTÓNICA]: Si se especifica sender_rol, el orquestador aborta la 
        transmisión hacia ese nodo específico, anulando el eco de red y previniendo la duplicación 
        de burbujas en interfaces de usuario optimistas (React).
        """
        if sesion_id in self.active_connections:
            for rol, connection in self.active_connections[sesion_id].items():
                if rol == sender_rol:
                    continue # Omisión algorítmica: El servidor ignora al emisor
                try:
                    await connection.send_text(json.dumps(message))
                except Exception as e:
                    logger.error(f"Fallo al enviar mensaje WebSocket a {rol} en sesión {sesion_id}: {str(e)}")

# Instancia global del gestor (Singleton). 
# Esta única variable manejará toda la carga de red en tiempo real de FastAPI.
manager = ConnectionManager()

const ConfigPanel = ({ isOpen, onClose }) => {
  const [vistaActiva, setVistaActiva] = useState('menu');
  const [urlPoliticas, setUrlPoliticas] = useState("");
  const [umbralRag, setUmbralRag] = useState(35.0);
  const [cargando, setCargando] = useState(false);
  const [mensaje, setMensaje] = useState({ tipo: '', texto: '' });

  useEffect(() => {
    if (isOpen) {
      fetch('http://localhost:8000/api/v1/admin/configuracion/global')
        .then(res => res.json())
        .then(data => {
          setUrlPoliticas(data.url_politicas);
          setUmbralRag(data.umbral_rag);
        })
        .catch(() => console.error("Error cargando configuración"));
    }
  }, [isOpen]);

  if (!isOpen) return null;

  const guardarConfiguracion = async (e) => {
    e.preventDefault();
    setCargando(true);
    setMensaje({ tipo: '', texto: '' });

    try {
      const response = await fetch('http://localhost:8000/api/v1/admin/configuracion/global', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ url_politicas: urlPoliticas, umbral_rag: parseFloat(umbralRag) }),
      });

      if (response.ok) {
        setMensaje({ tipo: 'exito', texto: 'Configuración actualizada.' });
      } else {
        setMensaje({ tipo: 'error', texto: 'Error al guardar la configuración.' });
      }
    } catch (error) {
      setMensaje({ tipo: 'error', texto: 'Fallo de conexión al servidor.' });
    } finally {
      setCargando(false);
    }
  };

  const renderMenu = () => (
    <div className="config-menu-lista">
      <div 
        className="config-menu-item" 
        onClick={() => { setVistaActiva('politicas'); setMensaje({ tipo: '', texto: '' }); }}

        <div className="config-item-text">
          <h5>Tratamiento de Datos</h5>
          <p>Configurar el enlace oficial de la universidad</p>
        </div>
        <div className="config-item-arrow">›</div>
      </div>
      
      <div 
        className="config-menu-item" 
        onClick={() => { setVistaActiva('similitud'); setMensaje({ tipo: '', texto: '' }); }}
      >
        <div className="config-item-icon">🎯</div>
        <div className="config-item-text">
          <h5>Precisión del Asistente</h5>
          <p>Nivel de similitud para respuestas automáticas</p>
        </div>
        <div className="config-item-arrow">›</div>
      </div>

      <div className="config-menu-item futuro">
        <div className="config-item-icon">🤖</div>
        <div className="config-item-text">
          <h5>Personalización del Agente</h5>
          <p>Ajustes de avatar, nombre y tono (Próximamente)</p>
        </div>
      </div>
    </div>
  );

  const renderPoliticas = () => (
    <div className="config-submodulo">
      <p className="config-descripcion">
        Define la URL oficial de Protección de Datos de la Universidad. El Widget redirigirá a los usuarios a este enlace automáticamente.
      </p>

      {mensaje.texto && (
        <div className={`config-alerta ${mensaje.tipo}`}>
          {mensaje.texto}
        </div>
      )}

      <form onSubmit={guardarConfiguracion} className="config-form">
        <label className="config-file-label">
          Enlace Institucional (URL)
          <input

            onChange={(e) => setUrlPoliticas(e.target.value)}
            placeholder="https://www.ucundinamarca.edu.co/..."
            required

          {cargando ? 'Guardando...' : 'Actualizar Enlace'}

  const renderSimilitud = () => (

      <p className="config-descripcion">
        Define el nivel de similitud mínimo requerido para que el asistente responda automáticamente. Si la certeza de la Inteligencia Artificial es menor a este porcentaje, se le ofrecerá al usuario la opción de hablar con un humano.
      </p>

      {mensaje.texto && (
        <div className={`config-alerta ${mensaje.tipo}`}>
          {mensaje.texto}
        </div>
      )}

      <form onSubmit={guardarConfiguracion} className="config-form">
        <label className="config-file-label">
          Similitud mínima requerida: {umbralRag}%
          <input 
            type="range" 
            min="10" 
            max="90" 
            step="1"

            onChange={(e) => setUmbralRag(e.target.value)} 
            style={{ marginTop: '10px' }}

          className="config-btn-submit"

  return (

      <div className="config-overlay" onClick={onClose}></div>

import React, { useState, useEffect, useRef, useCallback } from 'react';
import axios from 'axios';
import '../Dashboard.css';

const API_BASE_URL = "http://localhost:8000/api/v1/admin";
const WS_BASE_URL = "ws://localhost:8000/api/v1/chat";

const AdminBandejaPQR = () => {
  const [sesionesPendientes, setSesionesPendientes] = useState([]);
  
  // Persistencia de la sesión activa al cambiar de pestaña
  const [sesionActiva, setSesionActiva] = useState(() => {
    return localStorage.getItem('sesionPQRActiva') || null;
  });
  
  const [mensajes, setMensajes] = useState([]);
  const [textoInput, setTextoInput] = useState("");
  
  const [tipologia, setTipologia] = useState("");
  const [notaInterna, setNotaInterna] = useState("");
  
  const wsRef = useRef(null);
  const mensajesEndRef = useRef(null);

  const cargarSesionesPendientes = async () => {
    try {
      const response = await axios.get(`${API_BASE_URL}/sesiones/pendientes`);
      setSesionesPendientes(response.data);
    } catch (error) {
      console.error("Error obteniendo la cola de PQRs:", error);
    }
  };

  const conectarWebSocket = useCallback((sesionId) => {
    const ws = new WebSocket(`${WS_BASE_URL}/ws/${sesionId}/asesor`);
    ws.onmessage = (evento) => {
      const data = JSON.parse(evento.data);
      if (data.tipo === "nuevo_mensaje" || data.tipo === "alerta_sistema") {
        setMensajes(prev => [...prev, { emisor: data.emisor, contenido: data.contenido }]);
      }
    };
    wsRef.current = ws;
  }, []);

  const seleccionarSesion = useCallback(async (sesionId) => {
    if (wsRef.current) wsRef.current.close();
    setSesionActiva(sesionId);
    localStorage.setItem('sesionPQRActiva', sesionId);
    setMensajes([]);
    setTipologia(""); 
    setNotaInterna(""); 

    try {
      const response = await axios.get(`${API_BASE_URL}/sesiones/${sesionId}/historial`);
      setMensajes(response.data);
      conectarWebSocket(sesionId);
    } catch (error) {
      console.error("Error cargando historial:", error);
      if (error.response && error.response.status === 404) {
        setSesionActiva(null);
        localStorage.removeItem('sesionPQRActiva');
        setMensajes([]);
      }
    }
  }, [conectarWebSocket]);

  useEffect(() => {
    cargarSesionesPendientes();
    const intervalo = setInterval(cargarSesionesPendientes, 10000);
    return () => clearInterval(intervalo);
  }, []);

  useEffect(() => {
    if (sesionActiva) {
      seleccionarSesion(sesionActiva);
    }
  }, [sesionActiva, seleccionarSesion]);

  useEffect(() => {
    mensajesEndRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [mensajes]);

  const enviarMensaje = () => {
    if (!textoInput.trim() || !wsRef.current || wsRef.current.readyState !== WebSocket.OPEN) return;
    wsRef.current.send(JSON.stringify({ contenido: textoInput }));
    setMensajes(prev => [...prev, { emisor: 'asesor', contenido: textoInput }]);
    setTextoInput("");
  };

  const cerrarSesionPQR = async () => {
    if (!tipologia) {
      alert("Debes seleccionar una tipología antes de cerrar la PQR.");
      return;
    }
    if (!window.confirm("¿Deseas clausurar y resolver este ticket PQR?")) return;
    try {
      await axios.put(`${API_BASE_URL}/sesiones/${sesionActiva}/cerrar`, {
        tipologia: tipologia,
        nota_interna: notaInterna
      });
      if (wsRef.current) wsRef.current.close();
      setSesionActiva(null);
      localStorage.removeItem('sesionPQRActiva');
      setMensajes([]);
      setTipologia("");
      setNotaInterna("");
      cargarSesionesPendientes();
      alert("PQR cerrada con éxito.");
    } catch (error) {
      console.error("Error al cerrar sesión:", error);
    }
  };

  const sesionData = sesionesPendientes.find(s => s.sesion_id === sesionActiva);

  return (
    <div className="dashboard-container">
      {/* Columna Izquierda: Cola de Tickets */}
      <div className="sidebar-conversaciones">
        <div className="sidebar-header">
          Centro de Atención ({sesionesPendientes.length})
        </div>
        <div style={{ overflowY: 'auto', flex: 1 }}>
          {sesionesPendientes.length === 0 ? (
            <div style={{ padding: '20px', textAlign: 'center', color: '#888', fontSize: '0.85rem' }}>
              No hay PQRs en espera.
            </div>
          ) : (
            sesionesPendientes.map(sesion => (
              <div 
                key={sesion.sesion_id} 
                className={`sesion-item ${sesionActiva === sesion.sesion_id ? 'activa' : ''}`}
                onClick={() => seleccionarSesion(sesion.sesion_id)}
              >
                <div style={{ fontSize: '0.8rem', fontWeight: 'bold', color: '#333' }}>
                  {sesion.nombre_usuario || `Ticket #${sesion.sesion_id.substring(0, 8)}`}
                </div>
                <div style={{ fontSize: '0.75rem', color: '#666' }}>{sesion.sede} - {sesion.tipo_usuario}</div>

          )}
        </div>
      </div>

      {/* Columna Central: Chat en Vivo */}
      {!sesionActiva ? (

import React, { useState, useEffect, useRef } from 'react';
import '../Dashboard.css';

const AdminConocimiento = () => {
  // ---------------------------------------------------------

  const [modoInfinito, setModoInfinito] = useState(false);
  const [estadoTrabajo, setEstadoTrabajo] = useState('inactivo'); 
  const [jobId, setJobId] = useState(null);
  
  // LOGS GLOBALES
  const [logs, setLogs] = useState([]);
  
  // ---------------------------------------------------------
  // ESTADOS: AUTOMATIZACIÓN (FECHAS AGREGADAS)
  // ---------------------------------------------------------
  const [cronFrecuencia, setCronFrecuencia] = useState('diario');
  const [cronHora, setCronHora] = useState('02:00');
  const [cronDia, setCronDia] = useState('mon');
  const [cronFechaInicio, setCronFechaInicio] = useState('');
  const [cronFechaFin, setCronFechaFin] = useState('');
  const [programacionActiva, setProgramacionActiva] = useState(null);

  // ---------------------------------------------------------
  // ESTADOS: HISTORIAL DE ARCHIVOS (MODAL)
  // ---------------------------------------------------------
  const [modalAbierto, setModalAbierto] = useState(false);
  const [historialArchivos, setHistorialArchivos] = useState([]);
  const [cargandoHistorial, setCargandoHistorial] = useState(false);

  const consolaRef = useRef(null);

  // Auto-scroll de la consola
  useEffect(() => {
    if (consolaRef.current) {
      consolaRef.current.scrollTop = consolaRef.current.scrollHeight;
    }
  }, [logs]);

  // Carga inicial y Polling Global de Logs (Muestra la actividad CRON o manual)
  useEffect(() => {
    cargarProgramacionActual();
    
    const fetchLogs = async () => {
      try {
        const res = await fetch('http://localhost:8000/api/v1/admin/conocimiento/logs');
        if (res.ok) {
          const data = await res.json();
          setLogs(data.logs);
        }
      } catch (error) {
        // Ignorar fallos de red en polling de logs para no saturar consola
      }
    };
    
    fetchLogs();
    const intervalLogs = setInterval(fetchLogs, 3000);
    return () => clearInterval(intervalLogs);
  }, []);

  const cargarProgramacionActual = async () => {
    try {

      const data = await res.json();
      if (data.programado) {
        setProgramacionActiva(data);

        setProgramacionActiva(null);

      console.error("Error al cargar programación:", error);

    setCargandoHistorial(true);

      const res = await fetch('http://localhost:8000/api/v1/admin/conocimiento/historial-archivos');
      const data = await res.json();
      setHistorialArchivos(data.archivos || []);

      console.error("Error cargando archivos físicos:", error);
    } finally {
      setCargandoHistorial(false);
    }
  };

  const abrirModalHistorial = () => {
    setModalAbierto(true);
    cargarHistorialArchivos();
  };

  // Monitorear el status del job manual (Solo para habilitar botones y estado local)
  useEffect(() => {
    let intervaloConsulta;

    const verificarEstado = async () => {
      try {
        const respuesta = await fetch(`http://localhost:8000/api/v1/admin/conocimiento/estado/${jobId}`);
        if (!respuesta.ok) return;
        
        const datos = await respuesta.json();

        if (datos.estado_trabajo === 'completed') {
          setEstadoTrabajo('completado');
        } else if (datos.estado_trabajo === 'failed' || datos.estado_trabajo === 'error') {
          setEstadoTrabajo('error');
        } else if (datos.estado_trabajo === 'cancelled') {
          setEstadoTrabajo('cancelado');
        }

        setEstadoTrabajo('error');

            <button

            >

    setCargando(true);

    if (!datosArray) return [];
    const conteo = {};

      const val = item[campo] || "Sin Definir";
      conteo[val] = (conteo[val] || 0) + 1;

    if (!datosArray) return 0;

    let validos = 0;
    datosArray.forEach(item => {
      if(item.Fecha_Ingreso && item.Fecha_Actualizacion && item.Estado_Final === "Cerrada") {
         const inicio = new Date(item.Fecha_Ingreso);
         const fin = new Date(item.Fecha_Actualizacion);

    const filas = datos.datos_crudos.map(obj =>

    ).join("\n");

    const csvContent = "data:text/csv;charset=utf-8," + headers + "\n" + filas;
    const encodedUri = encodeURI(csvContent);
    const link = document.createElement("a");
    link.setAttribute("href", encodedUri);
    link.setAttribute("download", `Reporte_Analitico_PQR_${new Date().toISOString().split('T')[0]}.csv`);
    document.body.appendChild(link);

El agente automático institucional logró contener y resolver el ${datos.tasa_contencion}% de las solicitudes de manera autónoma, reduciendo drásticamente la carga operativa. Geográficamente, la sede con mayor demanda fue ${sedeMayor?.name || 'N/A'} concentrando ${sedeMayor?.value || 0} casos. El perfil predominante corresponde a ${rolMayor?.name || 'N/A'}. Requieren especial atención y seguimiento continuo los ${datos.escaladas_humano} casos que debieron ser escalados a talento humano por complejidad.`;

  const [sesionActiva, setSesionActiva] = useState(null);
  const [transcripcion, setTranscripcion] = useState([]);

  useEffect(() => {
    cargarHistorial();
  }, []);

  const cargarHistorial = async () => {
    setCargando(true);
    try {
      const response = await axios.get(`${API_BASE_URL}/historial-completo`);
      setSesiones(response.data);
    } catch (error) {
      console.error("Error al cargar historial general:", error);
    } finally {
      setCargando(false);
    }
  };

  const abrirConversacion = async (sesion) => {
    setSesionActiva(sesion);
    try {
      const res = await axios.get(`${API_BASE_URL}/sesiones/${sesion.id}/historial`);
      setTranscripcion(res.data);
    } catch (error) {
      console.error("No se pudo cargar la transcripción:", error);
      setTranscripcion([]);
    }
  };

  const calcularDuracion = (inicio, fin) => {
    const diff = (new Date(fin) - new Date(inicio)) / 60000; 
    return Math.max(diff, 1).toFixed(0); 
  };

  const sesionesFiltradas = sesiones.filter(s => {
    const cumpleBusqueda = s.nombre.toLowerCase().includes(busqueda.toLowerCase()) || 
                           s.documento.includes(busqueda) || 
                           s.correo.toLowerCase().includes(busqueda.toLowerCase());
    const cumpleFecha = filtroFecha ? s.creado.startsWith(filtroFecha) : true;
    return cumpleBusqueda && cumpleFecha;
  });

  return (
    <div className="estadisticas-container">
      <div style={{ backgroundColor: '#fff', borderRadius: '8px', padding: '1.5rem', boxShadow: '0 1px 3px rgba(0,0,0,0.05)', height: '100%', display: 'flex', flexDirection: 'column' }}>

              value={filtroFecha}
              onChange={(e) => setFiltroFecha(e.target.value)}
              style={{ padding: '8px 12px', border: '1px solid #ccc', borderRadius: '4px' }}
            />
            <input 
              type="text" 
              placeholder="Buscar por cédula, nombre o correo..." 
              value={busqueda}
              onChange={(e) => setBusqueda(e.target.value)}

            />

          </div>
        </div>

          {cargando ? (

          ) : (

import React, { useState, useEffect, useRef } from 'react';
import './Widget.css'; 

const Widget = () => {
  // ---------------------------------------------------------
  // Estados de Interfaz
  // ---------------------------------------------------------
  const [abierto, setAbierto] = useState(false);
  const [mostrarTooltip, setMostrarTooltip] = useState(false);
  
  // ---------------------------------------------------------
  // Estados Operativos
  // ---------------------------------------------------------
  const [sesionIniciada, setSesionIniciada] = useState(false);
  const [cargando, setCargando] = useState(false);
  const [sesionId, setSesionId] = useState("");
  
  const [mensajes, setMensajes] = useState([]);
  const [textoPregunta, setTextoPregunta] = useState("");
  const [escribiendo, setEscribiendo] = useState(false);
  const mensajesEndRef = useRef(null);
  const wsRef = useRef(null);

  // ---------------------------------------------------------
  // Datos Quemados para Pruebas Rápidas (Bypass de Tipeo)
  // ---------------------------------------------------------
  const [formData, setFormData] = useState({
    nombre: "Usuario de Prueba",
    documento: "100200300",
    correo: "estudiante.prueba@ucundinamarca.edu.co",
    celular: "3001234567",
    sede: "Fusagasugá",
    tipo_usuario: "Estudiante",
    politicas_aceptadas: true
  });

  // Animación suave del tooltip
  useEffect(() => {
    if (!abierto) {
      const timer = setTimeout(() => setMostrarTooltip(true), 3000);
      return () => clearTimeout(timer);
    } else {
      setMostrarTooltip(false);
    }
  }, [abierto]);

  useEffect(() => {
    mensajesEndRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [mensajes, sesionIniciada]);

  // RESTAURACIÓN DE BIDIRECCIONALIDAD Y ENRUTAMIENTO DIFY VÍA WEBSOCKET
  useEffect(() => {
    if (sesionIniciada && sesionId) {
      const ws = new WebSocket(`ws://localhost:8000/api/v1/chat/ws/${sesionId}/usuario`);
      ws.onmessage = (evento) => {
        const data = JSON.parse(evento.data);
        
        // Apagamos el indicador de escritura apenas recibimos cualquier respuesta
        setEscribiendo(false);

        if (data.tipo === "nuevo_mensaje" || data.tipo === "alerta_sistema") {
          setMensajes(prev => [...prev, { 
            emisor: data.emisor, 
            contenido: data.contenido,
            opciones: data.opciones 
          }]);
        }
      };
      wsRef.current = ws;
      return () => ws.close();
    }
  }, [sesionIniciada, sesionId]);

  const handleInputChange = (e) => {
    const { name, value, type, checked } = e.target;
    setFormData({
      ...formData,
      [name]: type === 'checkbox' ? checked : value
    });
  };

  // ---------------------------------------------------------
  // Inicialización (Endpoint POST de la Fase 1)
  // ---------------------------------------------------------
  const iniciarChat = async (e) => {
    e.preventDefault();
    setCargando(true);
    
    try {
      const respuesta = await fetch("http://localhost:8000/api/v1/chat/iniciar", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(formData)
      });
      
      const datos = await respuesta.json();
      
      if (respuesta.ok) {
        setSesionId(datos.sesion_id);
        setSesionIniciada(true);
        setMensajes([{
          emisor: "bot", 
          contenido: `¡Hola ${formData.nombre.split(' ')[0]}! Soy tu agente PQR de la Universidad de Cundinamarca. ¿En qué te puedo asesorar hoy?`
        }]);
      } else {
        alert("No pudimos iniciar la sesión: " + (datos.detail || "Verifica los campos."));
      }
    } catch (error) {
      console.error("Fallo de red:", error);
      alert("Error conectando con el servidor institucional.");
    } finally {
      setCargando(false);
    }
  };

  // ---------------------------------------------------------
  // Ciclo de Mensajería Unificado (Solo WebSocket)
  // ---------------------------------------------------------
  const enviarMensajeREST = async (e) => {
    e.preventDefault();
    if (!textoPregunta.trim()) return;

    const textoEnviado = textoPregunta;
    setTextoPregunta("");
    
    // Mostramos el mensaje en la pantalla del usuario inmediatamente
    setMensajes((prev) => [...prev, { emisor: "usuario", contenido: textoEnviado }]);

    // El backend se encarga ahora de rutear si va a Dify o al Humano
    if (wsRef.current && wsRef.current.readyState === WebSocket.OPEN) {
      setEscribiendo(true); // Encendemos el indicador de "Procesando tu solicitud..."
      wsRef.current.send(JSON.stringify({ contenido: textoEnviado }));
    } else {
      console.error("El WebSocket no está conectado.");
      setMensajes((prev) => [...prev, { emisor: "sistema", contenido: "Error de conexión en tiempo real. Intenta recargar la página." }]);
    }
  };

  const enviarOpcionWebSocket = (opcionValue) => {
    setTextoPregunta(opcionValue);
    setTimeout(() => document.querySelector('.udec-chat-input-form button').click(), 50);
  };

  return (
    <div className="udec-widget-root">
      
      {/* =========================================================
          BOTÓN FLOTANTE Y TOOLTIP
          ========================================================= */}
      {!abierto && (
        <div className="udec-launcher-container">
          {mostrarTooltip && (
            <div className="udec-tooltip">
              ¿Necesitas ayuda? Habla con nuestro agente
            </div>
          )}
          <button className="udec-launcher-btn" onClick={() => setAbierto(true)}>
            <svg viewBox="0 0 24 24" width="28" height="28" fill="currentColor">
              <path d="M12 2C6.48 2 2 6.48 2 12s4.48 10 10 10 10-4.48 10-10S17.52 2 12 2zm0 3c1.66 0 3 1.34 3 3s-1.34 3-3 3-3-1.34-3-3 1.34-3 3-3zm0 14.2c-2.5 0-4.71-1.28-6-3.22.03-1.99 4-3.08 6-3.08 1.99 0 5.97 1.09 6 3.08-1.29 1.94-3.5 3.22-6 3.22z"/>
            </svg>
          </button>
        </div>
      )}

      {/* =========================================================
          PANEL DEL CHAT ENCAPSULADO
          ========================================================= */}
      {abierto && (
        <div className="udec-chat-panel">
          
          <div className="udec-chat-header">
            <div className="udec-header-info">
              <span className="udec-header-avatar">🤖</span>
              <div>
                <h4 style={{margin: 0, fontSize: '0.95rem'}}>Agente PQR</h4>
                <p style={{margin: 0, fontSize: '0.7rem', opacity: 0.8}}>Universidad de Cundinamarca</p>
              </div>
            </div>
            <button className="udec-close-btn" onClick={() => setAbierto(false)}>✕</button>
          </div>

          <div className="udec-chat-body">
            
            {/* --- FASE PRE-CHAT: FORMULARIO INYECTADO EN LA BURBUJA --- */}
            {!sesionIniciada ? (
              <div className="udec-prechat-wrapper">
                <div className="udec-prechat-message">
                  <p>Para brindarte una orientación personalizada, por favor confírmanos tus datos institucionales.</p>
                </div>
                
                <form className="udec-form" onSubmit={iniciarChat}>
                  
                  {/* Ícono de Usuario */}
                  <div className="udec-input-group">
                    <span className="udec-input-icon">
                      <svg viewBox="0 0 24 24" width="20" height="20" fill="currentColor"><path d="M12 12c2.21 0 4-1.79 4-4s-1.79-4-4-4-4 1.79-4 4 1.79 4 4 4zm0 2c-2.67 0-8 1.34-8 4v2h16v-2c0-2.66-5.33-4-8-4z"/></svg>
                    </span>
                    <input type="text" name="nombre" value={formData.nombre} onChange={handleInputChange} placeholder="Nombre Completo" required />
                  </div>
                  
                  {/* Ícono de Identificación */}
                  <div className="udec-input-group">
                    <span className="udec-input-icon">
                      <svg viewBox="0 0 24 24" width="20" height="20" fill="currentColor"><path d="M21 3H3c-1.11 0-2 .89-2 2v14c0 1.11.89 2 2 2h18c1.11 0 2-.89 2-2V5c0-1.11-.89-2-2-2zm-9 15H4v-2h8v2zm0-4H4v-2h8v2zm6-4h-2V8h2v2zm2 4h-6V8h6v2z"/></svg>
                    </span>
                    <input type="text" name="documento" value={formData.documento} onChange={handleInputChange} placeholder="Cédula o NIT" required />
                  </div>

                  {/* Ícono de Correo */}
                  <div className="udec-input-group">
                    <span className="udec-input-icon">
                      <svg viewBox="0 0 24 24" width="20" height="20" fill="currentColor"><path d="M20 4H4c-1.1 0-1.99.9-1.99 2L2 18c0 1.1.9 2 2 2h16c1.1 0 2-.9 2-2V6c0-1.1-.9-2-2-2zm0 4l-8 5-8-5V6l8 5 8-5v2z"/></svg>
                    </span>
                    <input type="email" name="correo" value={formData.correo} onChange={handleInputChange} placeholder="Correo Electrónico" required />
                  </div>

                  {/* Ícono de Celular */}
                  <div className="udec-input-group">
                    <span className="udec-input-icon">
                      <svg viewBox="0 0 24 24" width="20" height="20" fill="currentColor"><path d="M17 1.01L7 1c-1.1 0-2 .9-2 2v18c0 1.1.9 2 2 2h10c1.1 0 2-.9 2-2V3c0-1.1-.9-1.99-2-1.99zM17 19H7V5h10v14z"/></svg>
                    </span>
                    <input type="tel" name="celular" value={formData.celular} onChange={handleInputChange} placeholder="Teléfono / Celular" required />
                  </div>
                  
                  {/* Ícono de Sede / Edificio */}
                  <div className="udec-input-group">
                    <span className="udec-input-icon">
                      <svg viewBox="0 0 24 24" width="20" height="20" fill="currentColor"><path d="M15 11V5l-3-3-3 3v2H3v14h18V11h-6zm-8 8H5v-2h2v2zm0-4H5v-2h2v2zm0-4H5V9h2v2zm6 8h-2v-2h2v2zm0-4h-2v-2h2v2zm0-4h-2V7h2v2zm0-4h-2V5h2v2zm6 12h-2v-2h2v2zm0-4h-2v-2h2v2z"/></svg>
                    </span>
                    <select name="sede" value={formData.sede} onChange={handleInputChange} required>
                      <option value="">Selecciona tu Sede...</option>
                      <option value="Fusagasugá">Fusagasugá</option>
                      <option value="Girardot">Girardot</option>
                      <option value="Ubaté">Ubaté</option>
                      <option value="Chía">Chía</option>
                      <option value="Facatativá">Facatativá</option>
                      <option value="Soacha">Soacha</option>
                      <option value="Zipaquirá">Zipaquirá</option>
                    </select>
                  </div>

                  {/* Ícono de Tipo de Usuario / Birrete */}
                  <div className="udec-input-group">
                    <span className="udec-input-icon">
                      <svg viewBox="0 0 24 24" width="20" height="20" fill="currentColor"><path d="M12 3L1 9l4 2.18v6L12 21l7-3.82v-6l2-1.09V17h2V9L12 3zm6.82 6L12 12.72 5.18 9 12 5.28 18.82 9zM17 15.99l-5 2.73-5-2.73v-3.72l5 2.73 5-2.73v3.72z"/></svg>
                    </span>
                    <select name="tipo_usuario" value={formData.tipo_usuario} onChange={handleInputChange} required>
                      <option value="">Soy...</option>
                      <option value="Estudiante">Estudiante</option>
                      <option value="Docente">Docente</option>
                      <option value="Administrativo">Administrativo</option>
                      <option value="Graduado">Graduado</option>
                      <option value="Ciudadano">Ciudadano</option>
                    </select>
                  </div>

                  <label className="udec-checkbox">
                    <input type="checkbox" name="politicas_aceptadas" checked={formData.politicas_aceptadas} onChange={handleInputChange} required />
                    <span>Acepto la <a href="http://localhost:8000/api/v1/admin/configuracion/politicas" target="_blank" rel="noopener noreferrer">política de tratamiento de datos</a>.</span>
                  </label>

                  <button type="submit" disabled={cargando} className="udec-submit-btn">
                    {cargando ? "Iniciando..." : "Comenzar Chat"}
                  </button>
                </form>
              </div>

            ) : (
              
              /* --- FASE IN-CHAT: MENSAJES Y CONVERSACIÓN --- */
              <div className="udec-chat-messages-container">
                <div className="udec-messages-scroll">
                  {mensajes.map((msg, idx) => (
                    <div key={idx} className={`udec-bubble-wrapper ${msg?.emisor?.includes('usuario') ? 'derecha' : 'izquierda'} ${msg.emisor === 'sistema' ? 'sistema' : ''}`}>
                      
                      {/* IDENTIDAD: Diferenciar Bot, Asesor y Usuario (Estilo WhatsApp) */}
                      {msg.emisor === 'asesor' && <div className="udec-asesor-tag">👤 Asesor Institucional</div>}
                      {msg.emisor === 'bot' && <div className="udec-asesor-tag" style={{color: '#2980b9'}}>🤖 Agente PQR</div>}
                      {msg.emisor === 'usuario' && <div className="udec-asesor-tag" style={{textAlign: 'right', color: '#555'}}>{formData.nombre.split(' ')[0]}</div>}

                      <div className="udec-bubble">
                        {msg.contenido}
                      </div>

                      {/* RENDERIZADO DE BOTONES DE OPCIÓN (Escalamiento interactivo) */}
                      {msg.opciones && (
                        <div className="widget-opciones-container">
                          {msg.opciones.map((opcion, i) => (
                            <button 
                              key={i} 
                              className="widget-opcion-btn"
                              onClick={() => enviarOpcionWebSocket(opcion.value)}
                            >
                              {opcion.label}
                            </button>
                          ))}
                        </div>
                      )}
                    </div>
                  ))}
                  
                  {/* INDICADOR DE ESCRITURA RESTAURADO (Sombreado Analizando) */}
                  {escribiendo && (
                    <div className="udec-bubble-wrapper izquierda">
                       <div className="udec-asesor-tag" style={{color: '#2980b9'}}>🤖 Agente PQR</div>
                       <div className="udec-bubble escribiendo-indicador">
                          Procesando tu solicitud...
                       </div>
                    </div>
                  )}

                  <div ref={mensajesEndRef} />
                </div>
                
                <form className="udec-chat-input-form" onSubmit={enviarMensajeREST}>
                  <input 
                    type="text" 
                    value={textoPregunta} 
                    onChange={(e) => setTextoPregunta(e.target.value)} 
                    placeholder="Escribe tu mensaje..." 
                  />
                  <button type="submit" disabled={!textoPregunta.trim()}>
                    <svg viewBox="0 0 24 24" width="20" height="20" fill="currentColor">
                      <path d="M2.01 21L23 12 2.01 3 2 10l15 2-15 2z"/>
                    </svg>
                  </button>
                </form>
              </div>
            )}

          </div>
        </div>
      )}
    </div>
  );
};

-- ==============================================================================
-- Esquema de Base de Datos Relacional - Agente PQR Universidad de Cundinamarca
-- Motor Objetivo - PostgreSQL 16
-- Descripción - Tablas operacionales para el orquestador cognitivo, manejo de sesiones,
-- historial de chat y escalamiento a humanos.
-- ==============================================================================

-- Habilitar extensión para generación de UUIDs nativos
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. Tabla de Asesores (Personal de la Universidad)
CREATE TABLE asesores (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(100) UNIQUE NOT NULL,
    rol VARCHAR(50) DEFAULT 'asesor', -- Valores permitidos 'asesor', 'coordinador', 'admin'
    estado VARCHAR(20) DEFAULT 'offline', -- Valores permitidos 'online', 'offline', 'ocupado'
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. Tabla de Sesiones (Interacciones de Estudiantes)
CREATE TABLE IF NOT EXISTS sesiones (
    id UUID PRIMARY KEY,
    estudiante_id VARCHAR(100),
    nombre_usuario VARCHAR(150),
    documento_identidad VARCHAR(50),
    correo VARCHAR(150),
    celular VARCHAR(50),
    sede VARCHAR(100),
    tipo_usuario VARCHAR(50),
    politicas_aceptadas BOOLEAN DEFAULT FALSE,
    estado VARCHAR(50) DEFAULT 'atencion_bot',
    asesor_asignado_id UUID REFERENCES asesores(id) ON DELETE SET NULL,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
-- 3. Tabla de Mensajes (Historial detallado del chat)
CREATE TABLE mensajes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    sesion_id UUID NOT NULL REFERENCES sesiones(id) ON DELETE CASCADE,
    emisor VARCHAR(20) NOT NULL, -- Valores permitidos 'estudiante', 'bot', 'asesor', 'sistema'
    contenido TEXT NOT NULL,
    similitud_rag NUMERIC(4, 3), -- Puntaje de similitud devuelto por Dify (ejemplo 0.852)
    url_fuente VARCHAR(500), -- Deep Link de donde Dify extrajo la información exacta
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Índices B-Tree para optimización de consultas de reportería y carga de historial en tiempo real
CREATE INDEX idx_mensajes_sesion ON mensajes(sesion_id);
CREATE INDEX idx_sesiones_estado ON sesiones(estado);

-- 4. Tabla de Tickets Diferidos (Gestión fuera de horario laboral)
CREATE TABLE tickets_pendientes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    sesion_id UUID UNIQUE NOT NULL REFERENCES sesiones(id) ON DELETE CASCADE,
    correo_contacto VARCHAR(100) NOT NULL,
    documento_identidad VARCHAR(50),
    resumen_problema TEXT NOT NULL,
    estado VARCHAR(20) DEFAULT 'abierto', -- Valores permitidos 'abierto', 'en_proceso', 'resuelto'
    asesor_resolucion_id UUID REFERENCES asesores(id) ON DELETE SET NULL,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    resuelto_en TIMESTAMP WITH TIME ZONE
);
